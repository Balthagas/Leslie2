\begin{definition}[Weak closure $sys^w$]
  \label{def:weakclosure}
  \lean{PLTS.System.weakClosure}\leanok
  \uses{def:weaktau, def:weakstep}
  The weak closure $sys^w$ keeps the states and the initial state of $sys$ and replaces each step by the weak transition of $sys$ out of the Dirac on its source: \leandecl{PLTS.System.weakClosure}.
\end{definition}
