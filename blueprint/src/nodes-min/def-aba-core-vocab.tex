\begin{definition}[The state one process carries between rounds]
  \label{def:aba-core-vocab}
  \lean{PLTS.ABA.ProcCore}\leanok
  The control state of one process in the round loop holds its input bit, its estimate, its round counter, its handshake phase, the round's graded outcome and a returned flag, and names no other process: \leandecl{PLTS.ABA.ProcCore}.
\end{definition}
