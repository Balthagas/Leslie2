\begin{theorem}[The layered presentation is exact]
  \label{thm:aba-layered-atd}
  \lean{PLTS.ABA.Layer.regG}\lean{PLTS.ABA.Layer.regC}\lean{PLTS.ABA.Layer.regA}\lean{PLTS.ABA.Layer.regroup}\lean{PLTS.ABA.Layer.layeredForward}\lean{PLTS.ABA.Layer.layeredConverse}\lean{PLTS.ABA.Layer.layeredSim}\lean{PLTS.ABA.Layer.layeredSimConverse}\lean{PLTS.ABA.Layer.layered_atd}\leanok

\end{theorem}
