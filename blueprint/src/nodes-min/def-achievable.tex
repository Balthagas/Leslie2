\begin{definition}[Achievable trace distributions]
  \label{def:achievable}
  \lean{PLTS.achievableTraceDists}\leanok
  \uses{def:traceprob}
  The achievable trace distributions $\operatorname{ATD}(sys)$ are the trace-probability functions of the probabilistic executions started from the Dirac on $sys.init$, and every refinement claim of this development is an inclusion of two such sets: \leandecl{PLTS.achievableTraceDists}.
\end{definition}
