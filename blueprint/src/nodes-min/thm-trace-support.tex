\begin{theorem}[Trace-support bridge]
  \label{thm:trace-support}
  \lean{PLTS.exists_exec_of_traceProb_ne_zero}\lean{PLTS.is_exec_of_probOf_ne_zero}\lean{PLTS.safety_transfer}\leanok

\end{theorem}
