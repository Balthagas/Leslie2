\begin{definition}[Distribution-monad lift $\mathcal{D}(sys)$]
  \label{def:dist}
  \lean{PLTS.System.dist}\leanok
  \uses{def:hyperstep}
  The distribution-monad lift $\mathcal{D}(sys)$ takes the distributions over states as its states and the hyper-steps of $sys$ as its steps, so that a distribution over states is a single state: \leandecl{PLTS.System.dist}.
\end{definition}
