\begin{theorem}[Substitution]
  \label{thm:aba-substitution}
  \lean{PLTS.ABA.substSim}\lean{PLTS.ABA.substitution}\leanok
  \uses{def:aba-layeredspec, def:aba-layered, thm:gbca-family-refines, thm:precong-parallel, thm:precong-abstract, thm:precong-relabel, thm:forward-sound, def:achievable}

\end{theorem}

  \uses{def:aba-layeredspec, def:aba-layered, thm:gbca-family-refines, thm:precong-parallel, thm:precong-abstract, thm:precong-relabel, thm:forward-sound}

\end{proof}
