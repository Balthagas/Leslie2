\begin{definition}[Probabilistic forward simulation]
  \label{def:forwardsim}
  \lean{PLTS.ProbabilisticForwardSimulation}\leanok
  \uses{def:weaktau, def:weakstep, def:pmfrel}
  A probabilistic forward simulation, the notion of Segala~\cite{Segala95}, relates each concrete state to a distribution over abstract states and answers every concrete step by a weak transition out of that distribution under a coupling: \leandecl{PLTS.ProbabilisticForwardSimulation}.
\end{definition}
