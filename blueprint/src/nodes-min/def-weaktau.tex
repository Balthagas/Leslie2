\begin{definition}[Silent weak transition]
  \label{def:weaktau}
  \lean{PLTS.weakTau}\leanok
  \uses{def:system}
  A silent weak transition $\mu_0 \Rightarrow_\tau \mu$ runs an internal-only scheduler from $\mu_0$ that halts almost surely with final-state distribution $\mu$: \leandecl{PLTS.weakTau}.
\end{definition}
