\begin{definition}[External weak transition]
  \label{def:weakstep}
  \lean{PLTS.weakStep}\leanok
  \uses{def:weaktau, def:hyperstep}
  An external weak transition is a $\tau^\ast\!\cdot l\cdot\tau^\ast$ chain, a silent weak transition followed by a hyper-step on $l$ followed by a second silent weak transition: \leandecl{PLTS.weakStep}.
\end{definition}
