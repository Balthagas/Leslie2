\begin{theorem}[The protocol under its composed reading]
  \label{thm:aba-protocol-composed}
  \lean{PLTS.ABA.ProtocolState}\lean{PLTS.ABA.ProtocolRel}\lean{PLTS.ABA.protocolRel_mk}\lean{PLTS.ABA.protocolRel_init}\lean{PLTS.ABA.protocolSim}\lean{PLTS.ABA.protocol_composed}\leanok
  \uses{def:aba-composed, def:aba-procnet, def:aba-netadv, def:aba-protocol, def:forwardsim, def:pmfrel, def:achievable, thm:forward-sound}
  The protocol is forward-simulated by its composed reading along the Dirac lift of $\mathrm{ProtocolRel}(P)$, the relation that pins of a composed state everything a state of the protocol determines in five unguarded conjuncts, and hence $\operatorname{ATD}(\mathrm{protocol}(P)) \subseteq \operatorname{ATD}(\mathrm{composed}(P))$; the inclusion is one-directional because a round instance answers the Byzantine graded-agreement drives, for which no program has a row (D11), and answers a send or a delivery at a process the protocol has terminated (D22): \leandecl{PLTS.ABA.protocolSim}, \leandecl{PLTS.ABA.protocol_composed}.
  A corrupted process's replaced program is matched loop for loop, and its corrupted return by the ABA-side network's Byzantine row (D23).
\end{theorem}

\begin{proof}\leanok
  \uses{def:aba-composed, def:aba-procnet, def:aba-netadv, def:aba-protocol, def:forwardsim, def:pmfrel, def:achievable, thm:forward-sound}
  See the Lean proof of \leandecl{PLTS.ABA.protocolSim}, then soundness (Theorem~\ref{thm:forward-sound}).
\end{proof}
