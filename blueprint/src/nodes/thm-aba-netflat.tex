\begin{theorem}[Safety of the deployed protocol]
  \label{thm:aba-netflat}
  \lean{PLTS.ABA.flatSpec_spec}\lean{PLTS.ABA.netFlat_spec}\lean{PLTS.ABA.netFlat_safe}\lean{PLTS.ABA.netFlat_traces}\lean{PLTS.ABA.exploded_safe}\leanok
  \uses{def:aba-procnet, def:aba-netadv, thm:aba-exploded-atd, thm:aba-substitution, thm:aba-flatspec-refines, thm:aba-coresim, thm:aba-spec-safe, thm:trace-support, thm:forward-sound, def:achievable}
  Let $P$ be parameters.
  Then
  $\operatorname{ATD}(\mathrm{netFlat}(P)) \subseteq
   \operatorname{ATD}(\mathrm{spec}(P))$,
  and every trace $t$ of positive probability under an achievable trace
  distribution of $\mathrm{netFlat}(P)$ satisfies Validity and Agreement.
  Such a $t$ also has positive probability under an achievable trace
  distribution of $\mathrm{flatSpec}(P)$.
  The same two conclusions hold of $\mathrm{exploded}(P)$.
  The statement carries no hypothesis on $t$ beyond positive probability.
  No corruption budget is assumed of the trace: the budget is a guard of the
  network adversary, and every execution of $\mathrm{netFlat}(P)$ respects it by
  construction.
\end{theorem}

\begin{proof}\leanok
  \uses{thm:aba-exploded-atd, thm:aba-substitution, thm:aba-flatspec-refines, thm:aba-coresim, thm:aba-spec-safe, thm:trace-support, thm:forward-sound, def:achievable}
  Theorem~\ref{thm:aba-substitution} gives
  $\operatorname{ATD}(\mathrm{netFlat}(P)) \subseteq
   \operatorname{ATD}(\mathrm{flatSpec}(P))$.
  This is the third conclusion.
  A trace distribution of $\mathrm{netFlat}(P)$ under which $t$ has positive
  probability is a trace distribution of $\mathrm{flatSpec}(P)$ under which it
  does.
  Theorem~\ref{thm:aba-flatspec-refines} and soundness
  (Theorem~\ref{thm:forward-sound}) applied to
  Theorem~\ref{thm:aba-coresim} give
  $\operatorname{ATD}(\mathrm{flatSpec}(P)) \subseteq
   \operatorname{ATD}(\mathrm{hybridSpec}(P)) \subseteq
   \operatorname{ATD}(\mathrm{spec}(P))$.
  Chaining the three inclusions by transitivity of set inclusion yields the
  first conclusion.
  By the trace-support bridge (Theorem~\ref{thm:trace-support}), safety of the
  specification's trace supports (Theorem~\ref{thm:aba-spec-safe}) transfers
  along that inclusion, which is the second.
  For the last, Theorem~\ref{thm:aba-exploded-atd} is an equality of the two
  sets of achievable trace distributions, so an achievable trace distribution of
  $\mathrm{exploded}(P)$ is one of $\mathrm{netFlat}(P)$ and the two conclusions
  read across.
\end{proof}
