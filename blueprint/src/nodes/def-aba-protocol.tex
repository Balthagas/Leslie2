\begin{definition}[The protocol]
  \label{def:aba-protocol}
  \lean{PLTS.System.mapIdle}\lean{PLTS.ABA.Net.wccPull}\lean{PLTS.ABA.Net.wccLift}\lean{PLTS.ABA.protocolPre}\lean{PLTS.ABA.protocolGroup}\lean{PLTS.ABA.protocol}\leanok
  \uses{def:aba-netadv, def:aba-procnet, def:aba-wcc-spec, def:aba-labels, def:relabel, def:syncproduct, def:process-algebra}
  The coin oracle $\mathrm{WCC.specFamily}(P)$ speaks $\mathrm{Lab}(n)$, so it is read over the extended alphabet by the construction $\mathrm{mapIdle}$ along the partial map $\mathrm{wccPull}$, which sends the two coin drives and the fused return $\mathrm{retWPub}$ to the coin handshake they stand for and every other rendezvous label to none: a label with an image steps by that image's rows, and a label without one is a self-loop.
  The protocol is built in three stages,
  \[
    \mathrm{protocolPre}(P) \;=\;
      \bigl(\mathrm{syncProduct}_{j \in [n]}\ \mathrm{ABAProcN}(P,j)\bigr)
        \parallel \bigl(\mathrm{netAdv}(P) \parallel \mathrm{wccLift}(P)\bigr),
  \]
  \[
    \mathrm{protocolGroup}(P) \;=\;
      \mathrm{relabel}\bigl(
        \mathrm{abstract}_{\mathrm{netEvtLabels}}(\mathrm{protocolPre}(P))\bigr),
  \]
  \[
    \mathrm{protocol}(P) \;=\; \mathrm{abstract}_H\bigl(\mathrm{protocolGroup}(P)\bigr),
  \]
  where $\mathrm{wccLift}(P)$ is the oracle read along $\mathrm{wccPull}$ and $H$ is the sub-protocol API of Definition~\ref{def:aba-labels}.
  Hiding $\mathrm{netEvtLabels}$ discharges the extended alphabet, so $\mathrm{protocolGroup}(P)$ speaks exactly $\mathrm{Lab}(n)$.
  Full construction: \leandecl{PLTS.ABA.protocol}.
\end{definition}
