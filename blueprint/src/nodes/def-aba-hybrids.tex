\begin{definition}[The analysis-side composition]
  \label{def:aba-hybrids}
  \lean{PLTS.ABA.context}\lean{PLTS.ABA.hybridSpec}\leanok
  \uses{def:aba-subspecs, def:aba-core, def:aba-labels, def:process-algebra, def:idle-family}
  The monolithic round loop of Definition~\ref{def:aba-core} beside the coin oracle, $\mathrm{context}(P) = \mathrm{core}(P) \parallel \mathrm{WCC.specFamily}(P)$, with the graded-agreement layer read at specification level as well:
  \[
    \mathrm{hybridSpec}(P) \;=\; \mathrm{abstract}_H\bigl(
      \mathrm{GBCA.specFamily}(P) \parallel \mathrm{context}(P) \bigr).
  \]
  The hidden alphabet $H$ is the sub-protocol API, the labels $\mathrm{callG}(r,i,b)$, $\mathrm{retG}(r,i,g)$, $\mathrm{callW}(r,i)$ and $\mathrm{retW}(r,i,c)$ for $r \in \mathbb{N}$, $i \in [n]$, bits $b$ and $c$, and graded outcomes $g \in \{A\,0, A\,1, B\,0, B\,1, C\}$, and the same set is hidden at the outer frame of $\mathrm{netFlat}(P)$, $\mathrm{exploded}(P)$ and $\mathrm{flatSpec}(P)$.
  Every system of the chain is therefore compared on the ABA interface alone, namely $\mathrm{callABA}(i,b)$, $\mathrm{retABA}(i,b)$, $\mathrm{fail}(i)$ and $\tau$.
  Each of the three layers carries its own copy of the corrupted set and applies the same budget-guarded transform to it on $\mathrm{fail}(k)$ (D1).
  Full construction: \leandecl{PLTS.ABA.hybridSpec}.
\end{definition}
