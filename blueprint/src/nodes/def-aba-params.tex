\begin{definition}[Parameters and the coin]
  \label{def:aba-params}
  \lean{PLTS.ABA.Params}\lean{PLTS.ABA.Params.wccPMF}\leanok
  The parameters: $n$ processes, at most $f$ corrupted with $3f < n$, coin goodness $\varepsilon$, and coin failure probability $\failmass$, subject to $2\varepsilon + \failmass \le 1$; the Lean field carrying $\failmass$ is named $\delta$.
  One distribution resolves both coins of the case study, that of rule~5 of Transition System~1 and that of the coin-resolution rule \leandeclnamed{flip}{PLTS.ABA.WCC.Step.flip}:
  \[
    \mathrm{wccPMF} \ =\ \varepsilon \cdot \delta(0) \ +\
      \varepsilon \cdot \delta(1) \ +\
      (1 - 2\varepsilon - \failmass) \cdot \delta(\top) \ +\
      \failmass \cdot \delta(\dagger),
  \]
  the constraint on the masses being what keeps the mass on $\top$ non-negative, and $\dagger$ the resolution that delivers to no process (D17).
  Full record: \leandecl{PLTS.ABA.Params}.
\end{definition}
