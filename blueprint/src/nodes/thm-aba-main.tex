\begin{theorem}[Refinement and correctness of ABA]
  \label{thm:aba-main}
  \lean{PLTS.ABA.refines}\lean{PLTS.ABA.main}\leanok
  \uses{thm:aba-protocol, thm:aba-protocol-composed, thm:aba-substitution, thm:aba-hybrid-spec, thm:aba-spec-safe}
  Every trace distribution achievable by the protocol is achievable by the ABA specification, and hence every positive-probability trace of the protocol satisfies Validity and Agreement, both in the form of Theorem~\ref{thm:aba-spec-safe} and so read at never-corrupted returners.
  The steps of the chain enter as inclusions chained by transitivity of set inclusion; transitivity of probabilistic forward simulation is not used.
  The chained simulation is additionally provided, along the composite of the three relations: the Dirac lift of $\mathrm{ProtocolRel}$, the pointwise round substitution, and the core relation.\lean{PLTS.ABA.chainSim}
  \leandeclnamed{refines}{PLTS.ABA.refines}, \leandeclnamed{main}{PLTS.ABA.main} and \leandeclnamed{chainSim}{PLTS.ABA.chainSim} are machine-checked axiom-clean, documented by pinned axiom guards.
\end{theorem}

\begin{proof}\leanok
  \uses{thm:aba-protocol, thm:aba-substitution, thm:aba-hybrid-spec, thm:aba-spec-safe, thm:trace-support, thm:forward-sound, thm:forward-trans}
  The two conclusions are those of Theorem~\ref{thm:aba-protocol}.
  For the chained simulation, apply transitivity of probabilistic forward simulation (Theorem~\ref{thm:forward-trans}) to the three simulations themselves rather than to their soundness inclusions.
  The two routes are independent, the inclusion never invoking transitivity of simulation, so the headline does not rest on it.
\end{proof}
