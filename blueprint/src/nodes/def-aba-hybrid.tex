\begin{definition}[The hybrid]
  \label{def:aba-hybrid}
  \lean{PLTS.ABA.HybridState}\lean{PLTS.ABA.hybridPre}\lean{PLTS.ABA.hybrid}\leanok
  \uses{def:aba-composed, def:aba-gbcasub, def:aba-gbca-spec, def:idle-family, def:relabel, def:process-algebra}
  The composed system of Definition~\ref{def:aba-composed} with its graded-agreement factor replaced by the family $\mathrm{specSide}(P)$ of round specifications, the other three factors and the whole pipeline unchanged:
  \[
    \mathrm{hybridPre}(P) \;=\;
      \mathrm{specSide}(P) \parallel \bigl(\mathrm{syncProduct}_{j}\ \mathrm{coreProcN}(P,j)
        \parallel (\mathrm{aNet}(P) \parallel \mathrm{wccLift}(P))\bigr),
  \]
  and $\mathrm{hybrid}(P)$ is that read through the two frames of $\mathrm{composed}(P)$.
  The system stays in the protocol's own coordinates: the round loops are $n$ programs, the DECIDED pools and the corrupted set are the ABA-side network's, and only the graded-agreement rounds are read at specification level.
  Full construction: \leandecl{PLTS.ABA.hybrid}.
\end{definition}
