% In this file you should put macros to be used only by
% the web version. Of course they should have a corresponding
% version in macros/print.tex.
% Typically the printed version could have more fancy decorations.
% This will probably be a very short file.

% The web edition carries no imager, so TikZ figures cannot render.
% Emit the caption text only; the figure file (#1) is never read on this path.
\newcommand{\blueprintfig}[2]{\begin{quote}\textit{Figure (print edition): #2}\end{quote}}

% A visible, doc-linked Lean declaration reference: typesets the full name,
% links it to the API docs, and registers it for checkdecls via \lean.
\newcommand{\leandecl}[1]{\href{https://sathiyavrs.github.io/Leslie2/docs/find/\#doc/#1}{\texttt{#1}}\lean{#1}}
% Same, with a short display name (arg 1) for a full declaration (arg 2) —
% used in rule tables where the qualified name would be too wide.
\newcommand{\leandeclnamed}[2]{\href{https://sathiyavrs.github.io/Leslie2/docs/find/\#doc/#2}{\texttt{#1}}\lean{#2}}

% Pointer to a whole Lean source file, by module path written with slashes and
% no extension, e.g. \leanmodule{Leslie2Protocols/ABA/Deployed}. That is the
% doc-gen4 page layout on disk: docs/Leslie2Protocols/ABA/Deployed.html.
\newcommand{\leanmodule}[1]{\href{https://sathiyavrs.github.io/Leslie2/docs/#1.html}{\texttt{#1}}}
