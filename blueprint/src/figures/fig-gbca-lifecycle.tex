% GBCA specification instance (PLTS.ABA.GBCA.Step). Automata style.
% Nodes are the five reachable (bind, grade) pairs; (bot, A) is unreachable
% because retA requires bind = some v. call/callLoop are enabled in every
% state and drawn once, on the initial state.
\begin{tikzpicture}[
    ->, >={Stealth[round]}, auto, semithick, on grid,
    every state/.style={draw, minimum size=1.3cm, align=center, font=\small},
    lbl/.style={font=\footnotesize, align=center, fill=white, inner sep=1.5pt}]

  \node[state]                           (bc) at (0,3.1)   {$(\bot,\mathrm C)$};
  \node[state]                           (vc) at (4.4,3.1) {$(v,\mathrm C)$};
  \node[state, initial, initial text={}] (bb) at (0,0)     {$(\bot,\bot)$};
  \node[state]                           (vb) at (4.4,0)   {$(v,\bot)$};
  \node[state]                           (va) at (8.8,0)   {$(v,\mathrm A)$};

  % self-loops
  \path
    (bb) edge[loop below, looseness=5]
      node[lbl, below]{1 \texttt{call}\\2 \texttt{callLoop}} (bb)
    (vb) edge[loop below, looseness=5]
      node[lbl, below]{4 \texttt{retB}} (vb)
    (va) edge[loop right, looseness=5]
      node[lbl, right]{4 \texttt{retB}\\5 \texttt{retA}} (va)
    (bc) edge[loop above, looseness=5]
      node[lbl, above]{6 \texttt{retC}} (bc)
    (vc) edge[loop above, looseness=5]
      node[lbl, above]{4 \texttt{retB}\\6 \texttt{retC}} (vc);

  % binding, the C-latch and the A-latch
  \path
    (bb) edge node[lbl, above]{3 \texttt{bindSet}} (vb)
    (bc) edge node[lbl, below]{3 \texttt{bindSet}} (vc)
    (vb) edge node[lbl, above]{5 \texttt{retA}} (va)
    (bb) edge node[lbl, left]{6 \texttt{retC}} (bc)
    (vb) edge node[lbl, right]{6 \texttt{retC}} (vc);
\end{tikzpicture}
